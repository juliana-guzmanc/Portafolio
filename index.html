// MotorMatch.gs — motor de compatibilidad de NEXT
function calcularMatch_(perfil, vacante, ctx) {
  const P = ctx.pesos;  // 40% hab · 25% exp · 20% ubic · 15% dispo
  const pHab = puntajeHabilidades_(perfil, vacante);
  const pExp = puntajeExperiencia_(perfil, vacante);
  const pUbi = puntajeUbicacion_(perfil, vacante);
  const pDis = puntajeDisponibilidad_(perfil, vacante);
  return Math.round(pHab*P.habilidades + pExp*P.experiencia + pUbi*P.ubicacion + pDis*P.disponibilidad);
}
